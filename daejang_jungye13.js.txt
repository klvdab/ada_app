const PORT = 9440;

/* ★2026-09-08 — 나가는 방송을 그대로 나스에 담아 둡니다(다시듣기용) */
const fs = require("fs");
/* ★2026-09-09 (빌드 260909-1) — ①방송 열 때 해설사 전화번호를 명단(live.php a=who)에 대조해 자격을 가립니다
   ②아이폰 사파리는 웹엠을 못 열므로 /mp3 창구에서 ffmpeg 로 소리만 mp3 로 바꿔 흘려보냅니다 */
const https = require("https");
const { spawn } = require("child_process");
const WHO_URL = "https://lvd.ada.or.kr/bfblive/live.php?a=who&phone=";
function haeseolsaChk(phone, cb){
  if (!phone) return cb(null);
  try {
    https.get(WHO_URL + encodeURIComponent(phone), (r) => {
      let d = ""; r.on("data", c => d += c);
      r.on("end", () => { try { const j = JSON.parse(d); cb(j && j.ok ? j : null); } catch(e) { cb(null); } });
    }).on("error", () => cb(null));
  } catch(e) { cb(null); }
}
function mp3Sijak(b){
  if (b.mp3) return b.mp3;
  const ff = spawn("/usr/bin/ffmpeg", ["-loglevel","error","-i","pipe:0","-vn","-c:a","libmp3lame","-ar","22050","-ac","1","-b:a","48k","-f","mp3","pipe:1"]);
  const m = { proc: ff, deutneun: new Set() };
  b.mp3 = m;
  ff.stdout.on("data", d => { for (const s of m.deutneun) { try { s.write(d); } catch(e) {} } });
  ff.on("close", () => { for (const s of m.deutneun) { try { s.end(); } catch(e) {} } if (b.mp3 === m) b.mp3 = null; });
  ff.stdin.on("error", () => {});
  ff.stderr.on("data", () => {});
  try { if (b.header) ff.stdin.write(b.header); } catch(e) {}
  return m;
}
function mp3Kkeut(b){
  if (!b.mp3) return;
  try { b.mp3.proc.stdin.end(); } catch(e) {}
  try { setTimeout(() => { try { b.mp3 && b.mp3.proc.kill(); } catch(e) {} }, 3000); } catch(e) {}
}
const SAVE = "/volume1/lvt_data/bfblive/sae";
function ppMk(){ try{ fs.mkdirSync(SAVE, { recursive:true }); }catch(e){} }
function extOf(mime){ return (mime||"").indexOf("mp4") >= 0 ? "mp4" : "webm"; }
function metaSsugi(sid, b, kkeutna){
  try{
    ppMk();
    const m = { sid:sid, title:b.title||"", who:b.who||"", jong:b.jong||"", chan:b.chan||"", mime:b.mime||"",
                ext:b.ext||"webm", sijak:Math.round(b.sijak/1000), jogak:b.jogak||0,
                bytes:b.bytes||0, cho:Math.round((b.maji-b.sijak)/1000),
                kkeut: kkeutna ? Math.round(Date.now()/1000) : 0 };
    fs.writeFileSync(SAVE + "/" + sid + ".json", JSON.stringify(m), "utf8");
  }catch(e){}
}
function jeojangSijak(sid, b){
  try{
    ppMk();
    b.ext = extOf(b.mime);
    b.ws = fs.createWriteStream(SAVE + "/" + sid + "." + b.ext, { flags:"w" });
    b.ws.on("error", function(){ b.ws = null; });
    metaSsugi(sid, b, false);
  }catch(e){ b.ws = null; }
}
function jeojangKkeut(sid, b){
  try{ if (b.ws) { b.ws.end(); b.ws = null; } }catch(e){}
  mp3Kkeut(b);
  metaSsugi(sid, b, true);
}

/* 방 하나 = 방송 하나 */
const bang = new Map();   /* sid → { tok, mime, header, jarik, deutneun:Set, sijak, maji } */

function jarikSae(){ return { tok:"", mime:"", header:null, deutneun:new Set(), sijak:Date.now(), maji:Date.now(), jogak:0, ws:null, bytes:0, ext:"webm", mp3:null, phone:"", chan:"" }; }

function json(res, obj, code){
  res.writeHead(code||200, {"Content-Type":"application/json; charset=utf-8","Cache-Control":"no-store","Access-Control-Allow-Origin":"*"});
  res.end(JSON.stringify(obj));
}

/* 길이를 앞에 붙여 보냅니다 — 받는 쪽이 조각 경계를 알 수 있게 */
function teulSsugi(buf){
  const head = Buffer.alloc(4);
  head.writeUInt32BE(buf.length, 0);
  return Buffer.concat([head, buf]);
}

const server = http.createServer((req, res) => {
  const u = new URL(req.url, "http://x");
  const p = u.pathname.replace(/\/+$/,"") || "/";
  const sid = u.searchParams.get("sid") || "";
  const tok = u.searchParams.get("tok") || "";

  if (req.method === "OPTIONS") { res.writeHead(204, {"Access-Control-Allow-Origin":"*","Access-Control-Allow-Headers":"*","Access-Control-Allow-Methods":"GET,POST,OPTIONS"}); return res.end(); }

  /* 살아 있는지 */
  if (p === "/sal") return json(res, { ok:true, pan:"1.3", bild:"260909-2", bang:bang.size });

  /* 지금 나가고 있는 것 목록 */
  if (p === "/moklok") {
    const rows = [];
    for (const [k, b] of bang) {
      rows.push({ sid:k, title:b.title || "", who:b.who || "", jong:b.jong || "", chan:b.chan || "", mime:b.mime,
                  deutneun:b.deutneun.size, cho: Math.round((Date.now()-b.sijak)/1000),
                  maji: Math.round((Date.now()-b.maji)/1000) });
    }
    rows.sort((x,y) => x.cho - y.cho);
    return json(res, { ok:true, rows });
  }

  /* 방송 열기 */
  if (p === "/yeolgi") {
    if (!sid) return json(res, { ok:false, msg:"방 번호가 없습니다." }, 400);
    const phone = (u.searchParams.get("phone") || "").replace(/[^0-9]/g, "");
    haeseolsaChk(phone, function(w){
      if (!w) return json(res, { ok:false, need:"phone", msg:"이 번호는 현장영상해설사 명단에 없습니다. 협회로 문의해 주십시오. 전화 02-363-4455" }, 403);
      const b = jarikSae();
      b.tok = tok || Math.random().toString(36).slice(2,12);
      b.mime = u.searchParams.get("mime") || "";
      b.title = (u.searchParams.get("title") || "").slice(0, 60);
      b.who = (w.name || u.searchParams.get("who") || "").slice(0, 30);
      b.jong = u.searchParams.get("jong") || "live";
      b.phone = phone;
      b.chan = (u.searchParams.get("chan") || "").slice(0, 30);   /* TV해설일 때 어느 채널인지 */
      bang.set(sid, b);
      jeojangSijak(sid, b);
      json(res, { ok:true, sid, tok:b.tok, mime:b.mime, who:b.who });
    });
    return;
  }

  /* 지금 방송 상태 */
  if (p === "/sangtae") {
    const b = bang.get(sid);
    if (!b) return json(res, { ok:true, on:false });
    return json(res, { ok:true, on:true, mime:b.mime, title:b.title||"", who:b.who||"", jong:b.jong||"", deutneun:b.deutneun.size, jogak:b.jogak,
                       cho: Math.round((Date.now()-b.sijak)/1000), maji: Math.round((Date.now()-b.maji)/1000) });
  }

  /* 조각 보내기 — 해설사 쪽 */
  if (p === "/jogak" && req.method === "POST") {
    const b = bang.get(sid);
    if (!b) return json(res, { ok:false, msg:"열린 방이 없습니다." }, 404);
    if (b.tok && tok !== b.tok) return json(res, { ok:false, msg:"열쇠가 맞지 않습니다." }, 403);
    const parts = [];
    req.on("data", d => parts.push(d));
    req.on("end", () => {
      const buf = Buffer.concat(parts);
      if (!buf.length) return json(res, { ok:true, skip:true });
      b.maji = Date.now(); b.jogak++;
      /* 첫 조각은 머리라 따로 두었다가, 나중에 들어오시는 분께 먼저 보냅니다 */
      if (!b.header) b.header = buf;
      try{ if (b.ws) b.ws.write(buf); b.bytes += buf.length; }catch(e){}
      try{ if (b.mp3) b.mp3.proc.stdin.write(buf); }catch(e){}
      if (b.jogak % 40 === 1) metaSsugi(sid, b, false);
      const out = teulSsugi(buf);
      for (const s of b.deutneun) { try { s.write(out); } catch(e) {} }
      json(res, { ok:true, n:b.jogak, deutneun:b.deutneun.size });
    });
    return;
  }

  /* 듣기 — 듣는 분 쪽. 끊지 않고 계속 흘려보냅니다 */
  if (p === "/deutgi") {
    const b = bang.get(sid);
    if (!b) return json(res, { ok:false, msg:"지금 나가는 방송이 없습니다." }, 404);
    res.writeHead(200, {
      "Content-Type":"application/octet-stream",
      "Cache-Control":"no-store, no-transform",
      "X-Accel-Buffering":"no",
      "Access-Control-Allow-Origin":"*"
    });
    if (b.header) { try { res.write(teulSsugi(b.header)); } catch(e) {} }
    b.deutneun.add(res);
    req.on("close", () => { b.deutneun.delete(res); });
    return;
  }

  /* 소리만 mp3 로 — 아이폰 사파리용 (2026-09-09) */
  if (p === "/mp3") {
    const b = bang.get(sid);
    if (!b) return json(res, { ok:false, msg:"지금 나가는 방송이 없습니다." }, 404);
    const m = mp3Sijak(b);
    res.writeHead(200, {
      "Content-Type":"audio/mpeg",
      "Cache-Control":"no-store, no-transform",
      "Accept-Ranges":"none",
      "X-Accel-Buffering":"no",
      "Access-Control-Allow-Origin":"*"
    });
    m.deutneun.add(res);
    req.on("close", () => { m.deutneun.delete(res); });
    return;
  }

  /* 방송 닫기 */
  if (p === "/datgi") {
    { const bb = bang.get(sid); if (bb) jeojangKkeut(sid, bb); }
    const b = bang.get(sid);
    if (b) { for (const s of b.deutneun) { try { s.end(); } catch(e) {} } bang.delete(sid); }
    return json(res, { ok:true });
  }

  /* 담아 둔 지난 방송 목록 */
  if (p === "/jinan") {
    let rows = [];
    try {
      rows = fs.readdirSync(SAVE).filter(function(f){ return f.slice(-5) === ".json"; })
        .map(function(f){ try { return JSON.parse(fs.readFileSync(SAVE + "/" + f, "utf8")); } catch(e) { return null; } })
        .filter(function(x){ return x; });
    } catch(e) {}
    rows.sort(function(a, b2){ return b2.sijak - a.sijak; });
    return json(res, { ok:true, rows:rows });
  }

  json(res, { ok:false, msg:"무엇을 하라는지 모르겠습니다." }, 404);
});

/* 오래 조용한 방은 스스로 치웁니다 */
setInterval(() => {
  const now = Date.now();
  for (const [sid, b] of bang) {
    if (now - b.maji > 10*60*1000) {
      jeojangKkeut(sid, b);
      for (const s of b.deutneun) { try { s.end(); } catch(e) {} }
      bang.delete(sid);
    }
  }
}, 60*1000);

server.headersTimeout = 0;
server.requestTimeout = 0;
server.timeout = 0;
server.keepAliveTimeout = 0;
server.listen(PORT, "0.0.0.0", () => { console.log("jungye listening " + PORT); });
